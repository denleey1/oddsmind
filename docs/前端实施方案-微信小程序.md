## 🔐 认证模块

### `utils/auth.js` - 认证工具类

```javascript
/**
 * 认证工具类
 * 负责微信登录、Token管理、用户状态管理
 */

const api = require('./api.js')
const storage = require('./storage.js')

// 存储键名
const TOKEN_KEY = 'jwt_token'
const USER_INFO_KEY = 'user_info'

class Auth {
  /**
   * 微信登录
   * @returns {Promise<Object>} 登录结果
   */
  async login() {
    try {
      // 1. 调用 wx.login() 获取 code
      const loginRes = await this.wxLogin()
      const code = loginRes.code

      // 2. 调用后端登录接口
      const response = await api.post('/miniprogram/login', { code })

      if (!response.success) {
        throw new Error(response.error || '登录失败')
      }

      const data = response.data

      // 3. 如果未激活，返回需要激活的状态
      if (!data.is_activated) {
        return {
          success: false,
          needActivate: true,
          openid: data.openid,
          message: data.message || '需要激活账号'
        }
      }

      // 4. 已激活，保存 Token 和用户信息
      if (data.token) {
        storage.set(TOKEN_KEY, data.token)
      }
      if (data.user) {
        storage.set(USER_INFO_KEY, data.user)
      }

      return {
        success: true,
        user: data.user,
        token: data.token
      }
    } catch (error) {
      console.error('登录失败:', error)
      return {
        success: false,
        error: error.message || '登录失败，请稍后重试'
      }
    }
  }

  /**
   * 激活账号（验证邀请码）
   * @param {string} inviteCode 邀请码
   * @returns {Promise<Object>} 激活结果
   */
  async activate(inviteCode) {
    try {
      // 1. 获取 code
      const loginRes = await this.wxLogin()
      const code = loginRes.code

      // 2. 调用激活接口
      const response = await api.post('/miniprogram/activate', {
        code,
        invite_code: inviteCode
      })

      if (!response.success) {
        throw new Error(response.error || '激活失败')
      }

      const data = response.data

      // 3. 保存 Token 和用户信息
      if (data.token) {
        storage.set(TOKEN_KEY, data.token)
      }
      if (data.user) {
        storage.set(USER_INFO_KEY, data.user)
      }

      return {
        success: true,
        user: data.user,
        token: data.token
      }
    } catch (error) {
      console.error('激活失败:', error)
      return {
        success: false,
        error: error.message || '激活失败，请检查邀请码是否正确'
      }
    }
  }

  /**
   * 获取当前用户信息
   * @returns {Promise<Object>} 用户信息
   */
  async getUserInfo() {
    try {
      const response = await api.get('/miniprogram/user')
      
      if (!response.success) {
        throw new Error(response.error || '获取用户信息失败')
      }

      // 更新本地存储的用户信息
      if (response.data) {
        storage.set(USER_INFO_KEY, response.data)
      }

      return {
        success: true,
        user: response.data
      }
    } catch (error) {
      console.error('获取用户信息失败:', error)
      return {
        success: false,
        error: error.message
      }
    }
  }

  /**
   * 检查登录状态
   * @returns {boolean} 是否已登录
   */
  isLoggedIn() {
    const token = storage.get(TOKEN_KEY)
    return !!token
  }

  /**
   * 检查是否已激活
   * @returns {boolean} 是否已激活
   */
  isActivated() {
    const userInfo = storage.get(USER_INFO_KEY)
    return userInfo && userInfo.is_activated === true
  }

  /**
   * 获取 Token
   * @returns {string|null} Token
   */
  getToken() {
    return storage.get(TOKEN_KEY)
  }

  /**
   * 获取用户信息
   * @returns {Object|null} 用户信息
   */
  getUserInfoSync() {
    return storage.get(USER_INFO_KEY)
  }

  /**
   * 登出
   */
  logout() {
    storage.remove(TOKEN_KEY)
    storage.remove(USER_INFO_KEY)
  }

  /**
   * 微信登录（封装 wx.login）
   * @returns {Promise<Object>} 登录结果
   */
  wxLogin() {
    return new Promise((resolve, reject) => {
      wx.login({
        success: resolve,
        fail: reject
      })
    })
  }

  /**
   * 检查账号是否过期
   * @returns {boolean} 是否过期
   */
  isExpired() {
    const userInfo = this.getUserInfoSync()
    if (!userInfo || !userInfo.expires_at) {
      return false
    }

    const expiresAt = new Date(userInfo.expires_at)
    return expiresAt < new Date()
  }
}

// 导出单例
module.exports = new Auth()
```

## 📱 登录页面

### `pages/login/login.js`

```javascript
const auth = require('../../utils/auth.js')

Page({
  data: {
    needActivate: false,        // 是否需要激活
    inviteCode: '',             // 邀请码
    loading: false,             // 加载状态
    errorMessage: ''            // 错误信息
  },

  onLoad(options) {
    // 如果已经登录且已激活，直接跳转
    if (auth.isLoggedIn() && auth.isActivated()) {
      this.redirectToHome()
      return
    }

    // 自动尝试登录
    this.autoLogin()
  },

  /**
   * 自动登录
   */
  async autoLogin() {
    this.setData({ loading: true, errorMessage: '' })

    try {
      const result = await auth.login()

      if (result.success) {
        // 登录成功，跳转到首页
        wx.showToast({
          title: '登录成功',
          icon: 'success'
        })
        setTimeout(() => {
          this.redirectToHome()
        }, 1500)
      } else if (result.needActivate) {
        // 需要激活
        this.setData({
          needActivate: true,
          loading: false
        })
      } else {
        // 登录失败
        this.setData({
          loading: false,
          errorMessage: result.error || '登录失败，请稍后重试'
        })
      }
    } catch (error) {
      this.setData({
        loading: false,
        errorMessage: error.message || '登录失败，请稍后重试'
      })
    }
  },

  /**
   * 输入邀请码
   */
  onInviteCodeInput(e) {
    this.setData({
      inviteCode: e.detail.value.trim().toUpperCase()
    })
  },

  /**
   * 提交激活
   */
  async onSubmitActivate() {
    const inviteCode = this.data.inviteCode.trim()

    if (!inviteCode) {
      wx.showToast({
        title: '请输入邀请码',
        icon: 'none'
      })
      return
    }

    this.setData({ loading: true, errorMessage: '' })

    try {
      const result = await auth.activate(inviteCode)

      if (result.success) {
        wx.showToast({
          title: '激活成功',
          icon: 'success'
        })
        setTimeout(() => {
          this.redirectToHome()
        }, 1500)
      } else {
        this.setData({
          loading: false,
          errorMessage: result.error || '激活失败，请检查邀请码'
        })
        wx.showToast({
          title: result.error || '激活失败',
          icon: 'none'
        })
      }
    } catch (error) {
      this.setData({
        loading: false,
        errorMessage: error.message || '激活失败，请稍后重试'
      })
      wx.showToast({
        title: '激活失败',
        icon: 'none'
      })
    }
  },

  /**
   * 重新登录
   */
  onRetryLogin() {
    this.setData({ needActivate: false })
    this.autoLogin()
  },

  /**
   * 跳转到首页
   */
  redirectToHome() {
    wx.switchTab({
      url: '/pages/index/index'  // 根据实际首页路径调整
    })
  }
})
```

## 🌐 API 请求封装

### `utils/api.js` - API 请求封装

```javascript
/**
 * API 请求封装
 * 自动添加 Token、处理 401、自动刷新 Token
 */

const auth = require('./auth.js')
const config = require('../config/config.js')

class Api {
  constructor() {
    this.baseURL = config.apiBaseURL  // 例如: 'https://your-domain.com/api'
  }

  /**
   * 通用请求方法
   */
  async request(options) {
    const {
      url,
      method = 'GET',
      data = {},
      header = {},
      needAuth = true  // 是否需要认证
    } = options

    // 构建完整 URL
    const fullURL = url.startsWith('http') ? url : `${this.baseURL}${url}`

    // 添加 Token
    if (needAuth) {
      const token = auth.getToken()
      if (token) {
        header['Authorization'] = `Bearer ${token}`
      }
    }

    // 设置默认 header
    header['Content-Type'] = header['Content-Type'] || 'application/json'

    return new Promise((resolve, reject) => {
      wx.request({
        url: fullURL,
        method: method,
        data: data,
        header: header,
        success: (res) => {
          // 处理 HTTP 状态码
          if (res.statusCode >= 200 && res.statusCode < 300) {
            resolve(res.data)
          } else if (res.statusCode === 401) {
            // Token 无效或过期，跳转到登录页
            this.handle401()
            reject(new Error('未登录或登录已过期'))
          } else {
            reject(new Error(res.data?.error || `请求失败: ${res.statusCode}`))
          }
        },
        fail: (err) => {
          console.error('请求失败:', err)
          reject(new Error('网络请求失败，请检查网络连接'))
        }
      })
    })
  }

  /**
   * GET 请求
   */
  async get(url, data = {}, options = {}) {
    return this.request({
      url,
      method: 'GET',
      data,
      ...options
    })
  }

  /**
   * POST 请求
   */
  async post(url, data = {}, options = {}) {
    return this.request({
      url,
      method: 'POST',
      data,
      ...options
    })
  }

  /**
   * PUT 请求
   */
  async put(url, data = {}, options = {}) {
    return this.request({
      url,
      method: 'PUT',
      data,
      ...options
    })
  }

  /**
   * DELETE 请求
   */
  async delete(url, data = {}, options = {}) {
    return this.request({
      url,
      method: 'DELETE',
      data,
      ...options
    })
  }

  /**
   * 处理 401 错误（未授权）
   */
  handle401() {
    // 清除本地 Token
    auth.logout()

    // 显示提示
    wx.showToast({
      title: '登录已过期，请重新登录',
      icon: 'none',
      duration: 2000
    })

    // 延迟跳转到登录页（避免与当前页面冲突）
    setTimeout(() => {
      wx.reLaunch({
        url: '/pages/login/login'
      })
    }, 2000)
  }
}

module.exports = new Api()
```

### `utils/storage.js` - 本地存储工具

```javascript
/**
 * 本地存储工具
 * 封装 wx.setStorageSync 和 wx.getStorageSync
 */

class Storage {
  /**
   * 设置存储
   */
  set(key, value) {
    try {
      wx.setStorageSync(key, value)
      return true
    } catch (error) {
      console.error('存储失败:', error)
      return false
    }
  }

  /**
   * 获取存储
   */
  get(key, defaultValue = null) {
    try {
      const value = wx.getStorageSync(key)
      return value !== '' ? value : defaultValue
    } catch (error) {
      console.error('获取存储失败:', error)
      return defaultValue
    }
  }

  /**
   * 删除存储
   */
  remove(key) {
    try {
      wx.removeStorageSync(key)
      return true
    } catch (error) {
      console.error('删除存储失败:', error)
      return false
    }
  }

  /**
   * 清空所有存储
   */
  clear() {
    try {
      wx.clearStorageSync()
      return true
    } catch (error) {
      console.error('清空存储失败:', error)
      return false
    }
  }
}

module.exports = new Storage()
```

## 🚀 启动逻辑

### `app.js` - 小程序入口

```javascript
const auth = require('./utils/auth.js')

App({
  onLaunch() {
    console.log('小程序启动')

    // 检查登录状态
    this.checkLoginStatus()
  },

  /**
   * 检查登录状态
   */
  async checkLoginStatus() {
    // 如果未登录，跳转到登录页
    if (!auth.isLoggedIn()) {
      console.log('未登录，跳转到登录页')
      wx.reLaunch({
        url: '/pages/login/login'
      })
      return
    }

    // 如果已登录但未激活，跳转到登录页（激活）
    if (!auth.isActivated()) {
      console.log('未激活，跳转到登录页')
      wx.reLaunch({
        url: '/pages/login/login'
      })
      return
    }

    // 检查账号是否过期
    if (auth.isExpired()) {
      console.log('账号已过期')
      wx.showModal({
        title: '提示',
        content: '您的账号已过期，请联系管理员',
        showCancel: false,
        success: () => {
          auth.logout()
          wx.reLaunch({
            url: '/pages/login/login'
          })
        }
      })
      return
    }

    // 尝试刷新用户信息
    try {
      await auth.getUserInfo()
    } catch (error) {
      console.error('获取用户信息失败:', error)
    }

    console.log('登录状态正常')
  },

  globalData: {
    userInfo: null
  }
})
```

## 👤 个人中心

### `pages/profile/profile.js`

```javascript
const auth = require('../../utils/auth.js')
const api = require('../../utils/api.js')

Page({
  data: {
    userInfo: null,
    isActivated: false,
    expiresAt: null,
    loading: true
  },

  onLoad() {
    this.loadUserInfo()
  },

  onShow() {
    // 每次显示页面时刷新用户信息
    this.loadUserInfo()
  },

  /**
   * 加载用户信息
   */
  async loadUserInfo() {
    this.setData({ loading: true })

    try {
      // 先从本地获取
      const localUserInfo = auth.getUserInfoSync()
      if (localUserInfo) {
        this.setData({
          userInfo: localUserInfo,
          isActivated: localUserInfo.is_activated || false,
          expiresAt: localUserInfo.expires_at,
          loading: false
        })
      }

      // 从服务器获取最新信息
      const result = await auth.getUserInfo()
      if (result.success && result.user) {
        this.setData({
          userInfo: result.user,
          isActivated: result.user.is_activated || false,
          expiresAt: result.user.expires_at,
          loading: false
        })
      }
    } catch (error) {
      console.error('加载用户信息失败:', error)
      this.setData({ loading: false })
    }
  },

  /**
   * 格式化过期时间
   */
  formatExpiresAt(expiresAt) {
    if (!expiresAt) return '永久有效'

    const date = new Date(expiresAt)
    const now = new Date()
    const diff = date - now

    if (diff < 0) {
      return '已过期'
    }

    const days = Math.floor(diff / (1000 * 60 * 60 * 24))
    if (days > 30) {
      return date.toLocaleDateString('zh-CN')
    } else if (days > 0) {
      return `${days}天后过期`
    } else {
      return '即将过期'
    }
  },

  /**
   * 登出
   */
  onLogout() {
    wx.showModal({
      title: '提示',
      content: '确定要退出登录吗？',
      success: (res) => {
        if (res.confirm) {
          auth.logout()
          wx.reLaunch({
            url: '/pages/login/login'
          })
        }
      }
    })
  }
})
```
