## 🎯 核心修改点

### 1. JS 文件修改

**文件**: `pages/match-detail/match-detail.js`

#### 添加 data 字段
```javascript
data: {
  // 新增
  poissonPrediction: null,
  poissonLoading: false,
  poissonError: null,
  poissonLoaded: false,
}
```

#### 修改 loadMatchDetail 方法
```javascript
async loadMatchDetail(matchId) {
  const response = await api.get(`/matches/${matchId}`)
  if (response.success && response.data) {
    const matchDetail = response.data
    // ✅ 删除 poissonPrediction（如果存在）
    if (matchDetail.poissonPrediction) {
      delete matchDetail.poissonPrediction
    }
    this.setData({ matchDetail: matchDetail })
  }
}
```

#### 添加 loadPoissonPrediction 方法
```javascript
async loadPoissonPrediction() {
  if (this.data.poissonLoaded || this.data.poissonLoading) return
  
  this.setData({ poissonLoading: true, poissonError: null })
  try {
    const response = await api.get(`/matches/${this.data.matchId}/poisson`)
    if (response.success && response.data) {
      this.setData({
        poissonPrediction: response.data,
        poissonLoaded: true,
        poissonLoading: false
      })
    }
  } catch (error) {
    this.setData({
      poissonLoading: false,
      poissonError: error.message
    })
  }
}
```

### 2. WXML 文件修改

**文件**: `pages/match-detail/match-detail.wxml`

```xml
<!-- 泊松预测区域 -->
<view class="poisson-section">
  <view class="section-title">泊松分布预测</view>
  
  <!-- 未加载 -->
  <view wx:if="{{!poissonLoaded && !poissonLoading}}">
    <button bindtap="loadPoissonPrediction" size="mini" type="primary">
      加载预测
    </button>
  </view>
  
  <!-- 加载中 -->
  <view wx:elif="{{poissonLoading}}">
    <text>加载中...</text>
  </view>
  
  <!-- 加载成功 -->
  <view wx:elif="{{poissonPrediction && !poissonPrediction.error}}">
    <!-- 显示预测内容 -->
    <view>预期进球: {{poissonPrediction.expected_goals.home}} vs {{poissonPrediction.expected_goals.away}}</view>
    <view>主胜: {{poissonPrediction.outcomes.home_win}}%</view>
    <view>平局: {{poissonPrediction.outcomes.draw}}%</view>
    <view>客胜: {{poissonPrediction.outcomes.away_win}}%</view>
  </view>
  
  <!-- 加载失败 -->
  <view wx:elif="{{poissonError}}">
    <text>{{poissonError}}</text>
    <button bindtap="loadPoissonPrediction" size="mini">重试</button>
  </view>
</view>
```
