## 📝 完整代码示例

### `utils/auth.js`（完整版）

```javascript
const api = require('./api.js')
const storage = require('./storage.js')

const TOKEN_KEY = 'jwt_token'
const USER_INFO_KEY = 'user_info'

class Auth {
  /**
   * 微信登录
   */
  async login() {
    // ... 登录逻辑
  }

  /**
   * 获取 Token
   * @returns {string|null}
   */
  getToken() {
    return storage.get(TOKEN_KEY)
  }

  /**
   * 获取用户信息
   * @returns {Object|null}
   */
  getUserInfoSync() {
    return storage.get(USER_INFO_KEY)
  }

  /**
   * 检查是否已登录
   * @returns {boolean}
   */
  isLoggedIn() {
    return !!this.getToken()
  }

  /**
   * 检查是否已激活
   * @returns {boolean}
   */
  isActivated() {
    const userInfo = this.getUserInfoSync()
    return userInfo && userInfo.is_activated === true
  }

  /**
   * 登出
   */
  logout() {
    storage.remove(TOKEN_KEY)
    storage.remove(USER_INFO_KEY)
  }

  /**
   * 微信登录（封装 wx.login）
   */
  wxLogin() {
    return new Promise((resolve, reject) => {
      wx.login({
        success: resolve,
        fail: reject
      })
    })
  }
}

// ✅ 导出实例（推荐）
module.exports = new Auth()
```

### `utils/api.js`（完整版）

```javascript
const auth = require('./auth.js')  // ✅ 导入 auth 实例

class Api {
  constructor() {
    this.baseURL = 'https://oddsmind.top'
  }

  /**
   * 通用请求方法
   */
  request(options = {}) {
    const {
      url,
      method = 'GET',
      data = {},
      header = {},
      needAuth = true
    } = options

    const fullURL = url.startsWith('http') ? url : `${this.baseURL}${url}`

    // ✅ 添加 Token
    if (needAuth) {
      const token = auth.getToken()  // ✅ 调用 getToken
      if (token) {
        header['Authorization'] = `Bearer ${token}`
      }
    }

    header['Content-Type'] = header['Content-Type'] || 'application/json'

    return new Promise((resolve, reject) => {
      wx.request({
        url: fullURL,
        method: method,
        data: data,
        header: header,
        success: (res) => {
          if (res.statusCode >= 200 && res.statusCode < 300) {
            resolve(res.data)
          } else if (res.statusCode === 401) {
            // Token 过期，清除本地数据
            auth.logout()
            reject(new Error('登录已过期，请重新登录'))
          } else {
            reject(new Error(res.data?.error || '请求失败'))
          }
        },
        fail: (err) => {
          reject(err)
        }
      })
    })
  }

  /**
   * GET 请求
   */
  get(url, data = {}, needAuth = true) {
    return this.request({
      url,
      method: 'GET',
      data,
      needAuth
    })
  }

  /**
   * POST 请求
   */
  post(url, data = {}, needAuth = true) {
    return this.request({
      url,
      method: 'POST',
      data,
      needAuth
    })
  }
}

// ✅ 导出实例
module.exports = new Api()
```
