# 前端激活后登录修复方案

## 🔍 问题分析

激活成功，但登录不进去。可能的原因：

1. **激活接口返回的数据格式问题**
2. **前端没有正确保存 token**
3. **激活成功后没有重新验证登录状态**

## ✅ 修复方案

### 修复1: 确保激活后正确保存数据

在 `utils/auth.js` 的 `activate()` 方法中添加详细日志和错误处理：

```javascript
async activate(inviteCode) {
  try {
    console.log('开始激活，邀请码:', inviteCode)
    
    // 1. 获取 code
    const loginRes = await this.wxLogin()
    const code = loginRes.code
    console.log('获取到微信code:', code ? '成功' : '失败')

    // 2. 调用激活接口
    console.log('调用激活接口...')
    const response = await api.post('/miniprogram/activate', {
      code,
      invite_code: inviteCode
    })
    
    console.log('激活接口响应:', response)
    console.log('响应success:', response.success)
    console.log('响应data:', response.data)

    if (!response.success) {
      console.error('激活失败:', response.error)
      throw new Error(response.error || '激活失败')
    }

    const data = response.data
    console.log('激活接口数据:', data)
    console.log('Token存在:', !!data.token)
    console.log('用户信息存在:', !!data.user)

    // 3. 保存 Token 和用户信息
    if (data.token) {
      storage.set(TOKEN_KEY, data.token)
      console.log('Token已保存，长度:', data.token.length)
    } else {
      console.error('激活接口未返回token!')
      throw new Error('激活失败：未返回token')
    }

    if (data.user) {
      storage.set(USER_INFO_KEY, data.user)
      console.log('用户信息已保存:', data.user)
    } else {
      console.error('激活接口未返回用户信息!')
      throw new Error('激活失败：未返回用户信息')
    }

    // 4. 验证保存是否成功
    const savedToken = storage.get(TOKEN_KEY)
    const savedUser = storage.get(USER_INFO_KEY)
    console.log('验证保存结果 - Token:', savedToken ? '已保存' : '未保存')
    console.log('验证保存结果 - 用户信息:', savedUser ? '已保存' : '未保存')

    return {
      success: true,
      user: data.user,
      token: data.token
    }
  } catch (error) {
    console.error('激活失败:', error)
    return {
      success: false,
      error: error.message || '激活失败，请检查邀请码是否正确'
    }
  }
}
```

### 修复2: 激活成功后重新登录

在 `pages/login/login.js` 中，激活成功后重新调用登录接口：

```javascript
async onSubmitActivate() {
  const inviteCode = this.data.inviteCode.trim()

  if (!inviteCode) {
    wx.showToast({
      title: '请输入邀请码',
      icon: 'none'
    })
    return
  }

  this.setData({ loading: true, errorMessage: '' })

  try {
    console.log('开始激活，邀请码:', inviteCode)
    const result = await auth.activate(inviteCode)
    console.log('激活结果:', result)

    if (result.success) {
      // ✅ 验证 token 是否已保存
      const token = auth.getToken()
      const userInfo = auth.getUserInfoSync()
      
      console.log('激活后验证 - Token:', token ? '已保存' : '未保存')
      console.log('激活后验证 - 用户信息:', userInfo)
      console.log('激活后验证 - 是否激活:', userInfo?.is_activated)

      if (!token || !userInfo) {
        console.error('激活后数据未保存，重新登录')
        // 数据未保存，重新登录
        setTimeout(() => {
          this.autoLogin()
        }, 1000)
        return
      }

      wx.showToast({
        title: '激活成功',
        icon: 'success'
      })
      
      // ✅ 激活成功后，重新调用登录接口，确保获取最新的token
      setTimeout(async () => {
        try {
          console.log('激活成功，重新登录...')
          const loginResult = await auth.login()
          console.log('重新登录结果:', loginResult)
          
          if (loginResult.success) {
            console.log('重新登录成功，跳转首页')
            this.redirectToHome()
          } else {
            console.error('重新登录失败:', loginResult.error)
            // 如果重新登录失败，但token已保存，直接跳转
            if (auth.isLoggedIn() && auth.isActivated()) {
              console.log('Token已保存，直接跳转')
              this.redirectToHome()
            } else {
              wx.showToast({
                title: loginResult.error || '登录失败，请重试',
                icon: 'none'
              })
            }
          }
        } catch (error) {
          console.error('重新登录异常:', error)
          // 如果重新登录失败，但token已保存，直接跳转
          if (auth.isLoggedIn() && auth.isActivated()) {
            console.log('Token已保存，直接跳转')
            this.redirectToHome()
          } else {
            wx.showToast({
              title: '登录失败，请重试',
              icon: 'none'
            })
          }
        }
      }, 1500)
    } else {
      this.setData({
        loading: false,
        errorMessage: result.error || '激活失败，请检查邀请码'
      })
      wx.showToast({
        title: result.error || '激活失败',
        icon: 'none'
      })
    }
  } catch (error) {
    console.error('激活异常:', error)
    this.setData({
      loading: false,
      errorMessage: error.message || '激活失败，请稍后重试'
    })
    wx.showToast({
      title: '激活失败',
      icon: 'none'
    })
  }
}
```

### 修复3: 检查首页登录状态

在首页的 `onLoad()` 中，添加详细的登录检查：

```javascript
// pages/index/index.js
onLoad() {
  console.log('首页加载，检查登录状态...')
  
  // ✅ 检查登录状态
  const token = auth.getToken()
  const userInfo = auth.getUserInfoSync()
  
  console.log('首页 - Token:', token ? '存在' : '不存在')
  console.log('首页 - 用户信息:', userInfo)
  console.log('首页 - 是否激活:', userInfo?.is_activated)
  
  if (!token) {
    console.log('未找到Token，跳转到登录页')
    wx.redirectTo({
      url: '/pages/login/login'
    })
    return
  }
  
  if (!userInfo) {
    console.log('未找到用户信息，跳转到登录页')
    wx.redirectTo({
      url: '/pages/login/login'
    })
    return
  }
  
  if (!userInfo.is_activated) {
    console.log('用户未激活，跳转到登录页')
    wx.redirectTo({
      url: '/pages/login/login'
    })
    return
  }
  
  console.log('登录状态验证通过，加载数据')
  // 已登录，加载数据
  this.loadData()
}
```

## 🔍 常见问题

### Q1: 激活成功后，为什么还是显示未激活？

**A:** 可能的原因：
1. 用户信息未正确保存
2. 前端状态未更新
3. 数据库中的状态未更新

**解决：** 检查存储中的用户信息，确保 `is_activated` 为 `true`

### Q2: Token 保存了，但为什么还是登录不进去？

**A:** 可能的原因：
1. Token 格式不正确
2. Token 已过期
3. 首页登录检查逻辑有问题

**解决：** 检查 token 格式，确保是字符串；检查首页登录检查逻辑

### Q3: 如何验证 Token 是否有效？

**A:** 可以调用用户信息接口：

```javascript
const api = require('../../utils/api.js')

async function testToken() {
  try {
    const response = await api.get('/miniprogram/user')
    console.log('Token有效，用户信息:', response)
    return true
  } catch (error) {
    console.error('Token无效:', error)
    return false
  }
}
```
